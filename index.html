# Anaplan REST API — Automated P&L Data Loader
# Automates monthly actuals upload & import action trigger
# Use case: FP&A Budgeting & Forecasting (Automotive Mfg.)

import requests, pandas as pd
from base64 import b64encode
from datetime import datetime

class AnaplanConnector:
    """Handles auth, file upload & action execution via Anaplan REST API."""

    BASE_URL = "https://api.anaplan.com/2/0/workspaces"

    def __init__(self, workspace_id, model_id, username, password):
        self.workspace_id = workspace_id
        self.model_id     = model_id
        self.session      = requests.Session()
        self.session.headers["Authorization"] = "Basic " + b64encode(
            f"{username}:{password}".encode()
        ).decode()

    def upload_file(self, file_id: str, df: pd.DataFrame) -> bool:
        """Upload a Pandas DataFrame as CSV to an Anaplan file."""
        csv_bytes = df.to_csv(index=False).encode("utf-8")
        url = (
            f"{self.BASE_URL}/{self.workspace_id}/models/"
            f"{self.model_id}/files/{file_id}"
        )
        resp = self.session.put(
            url, data=csv_bytes,
            headers={"Content-Type": "application/octet-stream"}
        )
        return resp.status_code == 200

    def run_import(self, action_id: str) -> str:
        """Trigger an Anaplan import action and return task state."""
        url = (
            f"{self.BASE_URL}/{self.workspace_id}/models/"
            f"{self.model_id}/actions/{action_id}/tasks"
        )
        resp = self.session.post(url, json={"localeName": "en_US"})
        return resp.json()["task"]["taskState"]
